def play_tone(freq, duration=0.4, volume=0.3):
    if not AUDIO_AVAILABLE: return
    try:
        n_samples = int(AUDIO_FREQ * duration)
        buf = array.array('h', [0] * n_samples)
        max_amp = 32767 * volume
        for i in range(n_samples):
            t = float(i) / AUDIO_FREQ
            envelope = max(0.0, 1.0 - (i / n_samples))
            val = int(math.sin(2.0 * math.pi * freq * t) * max_amp * envelope)
            buf[i] = val
        sound = pygame.mixer.Sound(buffer=buf)
        sound.play()
    except Exception: pass

class NoteExplosion(pygame.sprite.Sprite):
    def __init__(self, x, y, color):
        super().__init__()
        self.particles = []
        for _ in range(12):
            angle = random.uniform(0, math.pi * 2)
            speed = random.uniform(2, 6)
            self.particles.append([x, y, math.cos(angle) * speed, math.sin(angle) * speed, random.randint(12, 20)])
        self.color = color
        self.image = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA)
        self.rect = self.image.get_rect()

    def update(self):
        alive = False
        self.image.fill((0, 0, 0, 0))
        for p in self.particles:
            if p[4] > 0:
                p[0] += p[2]
                p[1] += p[3]
                p[4] -= 1
                alive = True
                pygame.draw.circle(self.image, self.color, (int(p[0]), int(p[1])), max(1, p[4] // 3))
        if not alive:
            self.kill()

class SpiderEnemy(pygame.sprite.Sprite):
    def __init__(self, target_player, difficulty_factor):
        super().__init__()
        self.target = target_player
        self.image = pygame.Surface((46, 36), pygame.SRCALPHA)
        self.rect = self.image.get_rect(x=WIDTH + random.randint(20, 100), y=GROUND_Y - 36)
        self.base_speed = (random.uniform(1.5, 2.6) + difficulty_factor * 0.5) * 1.05
        self.wobble_timer = random.uniform(0, math.pi * 2)
        self._draw_spider()

    def _draw_spider(self):
        pygame.draw.ellipse(self.image, COLOR_SPIDER, (10, 8, 26, 20))
        for px in range(12, 34, 4):
            pygame.draw.line(self.image, (40, 0, 50), (px, 6), (px, 10), 2)
        for ox in [14, 18, 22, 26]:
            pygame.draw.circle(self.image, (255, 50, 0), (ox, 13), 2)
            pygame.draw.circle(self.image, (255, 255, 255), (ox, 13), 1)
        pygame.draw.lines(self.image, COLOR_SPIDER, False, [(12, 18), (2, 4), (0, 24)], 3)
        pygame.draw.lines(self.image, COLOR_SPIDER, False, [(32, 18), (42, 4), (44, 24)], 3)

    def update(self):
        self.wobble_timer += 0.12
        dx = self.target.rect.x - self.rect.x
        dy = self.target.rect.y - self.rect.y
        dist = math.hypot(dx, dy)
        if dist != 0:
            self.rect.x += (dx / dist) * self.base_speed
            self.rect.y += (dy / dist) * (self.base_speed * 0.4) + math.sin(self.wobble_timer) * 1.5
        if self.rect.bottom >= GROUND_Y:
            self.rect.bottom = GROUND_Y
        if self.rect.right < 0:
            self.kill()

def run_level():
    timmy = Timmy()
    bg_art = NightViolinBackground()
    all_sprites = pygame.sprite.Group(timmy)
    projectiles = pygame.sprite.Group()
    fireballs = pygame.sprite.Group()
    items = pygame.sprite.Group()
    enemies = pygame.sprite.Group()
    clef_boxes = pygame.sprite.Group()
    silence_bombs = pygame.sprite.Group()
    explosions = pygame.sprite.Group()
    keys_group = pygame.sprite.Group()
    doors_group = pygame.sprite.Group()

    start_ticks = pygame.time.get_ticks()
    scroll_offset = 0
    boss_spawned = False
    boss_ref = None
    game_over = False
    victory = False
    is_paused = False
    door_spawned = False

    spawn_enemy_timer = 0
    spawn_giant_timer = 0
    spawn_item_timer = 0
    spawn_bomb_timer = 0
    note_in_song_idx = 0

    font_hud = pygame.font.Font(None, 26)
    font_boss = pygame.font.Font(None, 34)
    font_big = pygame.font.Font(None, 54)
    font_huge_baal = pygame.font.Font(None, 68)
    font_btn = pygame.font.Font(None, 32)

    baal_title_timer = 0

    running = True
    while running:
        clock.tick(FPS)
        scroll_offset += 1.5

        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                return "QUIT"
            elif event.type == pygame.KEYDOWN:
                if event.key == pygame.K_ESCAPE:
                    return "QUIT"
                if event.key == pygame.K_p:
                    is_paused = not is_paused
                if game_over or victory:
                    if event.key == pygame.K_r:
                        return "RESTART"
                    elif event.key == pygame.K_q:
                        return "QUIT"
                if not game_over and not victory and not is_paused:
                    if event.key in (pygame.K_SPACE, pygame.K_w, pygame.K_UP):
                        timmy.jump()
                    elif event.key in (pygame.K_s, pygame.K_DOWN):
                        timmy.crouch(True)
                    elif event.key in (pygame.K_j, pygame.K_z):
                        if timmy.ammo_sharps > 0:
                            p = Projectile(timmy.rect.right, timmy.rect.centery, "SHARP", timmy.weapon_level)
                            all_sprites.add(p)
                            projectiles.add(p)
                            timmy.ammo_sharps -= 1
                            play_white_noise_shot()
                    elif event.key in (pygame.K_k, pygame.K_x):
                        if timmy.ammo_flats > 0:
                            p = Projectile(timmy.rect.right, timmy.rect.centery, "FLAT_GRANADE")
                            all_sprites.add(p)
                            projectiles.add(p)
                            timmy.ammo_flats -= 1
                            play_white_noise_shot()
            elif event.type == pygame.KEYUP:
                if event.key in (pygame.K_s, pygame.K_DOWN):
                    timmy.crouch(False)

        if not game_over and not victory and not is_paused:
            elapsed_sec = (pygame.time.get_ticks() - start_ticks) / 1000
            difficulty_factor = min(1.0, elapsed_sec / GAME_DURATION_SEC)

            if not boss_spawned:
                spawn_enemy_timer += 1
                spawn_giant_timer += 1
                spawn_item_timer += 1
                spawn_bomb_timer += 1

                if spawn_enemy_timer >= max(20, int(50 - (difficulty_factor * 20))):
                    spider = SpiderEnemy(timmy, difficulty_factor)
                    all_sprites.add(spider)
                    enemies.add(spider)
                    spawn_enemy_timer = 0

                if spawn_giant_timer >= 420:
                    g_spider = GiantSpiderEnemy(fireballs)
                    all_sprites.add(g_spider)
                    enemies.add(g_spider)
                    spawn_giant_timer = 0

                bomb_interval = max(80, int(200 - (difficulty_factor * 70)))
                if spawn_bomb_timer >= bomb_interval:
                    s_bomb = SilenceBomb(difficulty_factor)
                    all_sprites.add(s_bomb)
                    silence_bombs.add(s_bomb)
                    spawn_bomb_timer = 0

                if spawn_item_timer >= 80:
                    note_key, figure_type = PARA_ELISE_SEQUENCE[note_in_song_idx]
                    note_in_song_idx += 1
                    if note_in_song_idx >= len(PARA_ELISE_SEQUENCE):
                        note_in_song_idx = 0

                    note = ItemNote(note_key, figure_type, difficulty_factor)
                    all_sprites.add(note)
                    items.add(note)

                    if random.random() < 0.4:
                        clef = ClefBox()
                        all_sprites.add(clef)
                        clef_boxes.add(clef)
                    spawn_item_timer = 0

                if elapsed_sec >= GAME_DURATION_SEC and not boss_spawned:
                    boss_spawned = True
                    boss_ref = BossBaal(fireballs)
                    all_sprites.add(boss_ref)
                    enemies.add(boss_ref)
                    baal_title_timer = 120

            if baal_title_timer > 0:
                baal_title_timer -= 1

            all_sprites.update()
            silence_bombs.update()
            fireballs.update()
            explosions.update()
            keys_group.update()
            doors_group.update()

            for note in items:
                if timmy.rect.colliderect(note.rect):
                    play_tone(note.freq, note.sound_duration, 0.4)
                    timmy.score_times += note.value
                    if note.type == "REDONDA":
                        timmy.lives = min(25, timmy.lives + 3)
                    elif note.type == "BLANCA":
                        timmy.lives = min(25, timmy.lives + 1)

                    timmy.evolve_weapon_check()
                    exp = NoteExplosion(note.rect.centerx, note.rect.centery, note.color)
                    all_sprites.add(exp)
                    note.kill()

            for clef in clef_boxes:
                if timmy.rect.colliderect(clef.rect):
                    timmy.ammo_sharps += 25
                    timmy.ammo_flats += 10
                    play_powerup_sound()
                    clef.kill()

            for bomb in silence_bombs:
                if timmy.rect.colliderect(bomb.rect):
                    timmy.lives -= 3
                    play_bomb_explosion_sound()
                    exp = MassiveExplosion(bomb.rect.centerx, bomb.rect.centery, 50)
                    all_sprites.add(exp)
                    if timmy.lives <= 0:
                        game_over = True
                        play_defeat_sound()
                    bomb.kill()

            for p in projectiles:
                for bomb in pygame.sprite.spritecollide(p, silence_bombs, True):
                    bg_art.trigger_surprise()
                    play_bomb_explosion_sound()
                    exp = MassiveExplosion(bomb.rect.centerx, bomb.rect.centery, 40)
                    all_sprites.add(exp)
                    p.kill()

                for enemy in pygame.sprite.spritecollide(p, enemies, False):
                    bg_art.trigger_surprise()
                    if isinstance(enemy, GiantSpiderEnemy) or isinstance(enemy, BossBaal):
                        enemy.hp -= 1
                        play_big_bug_death()
                        if enemy.hp <= 0:
                            play_bomb_explosion_sound()
                            exp = MassiveExplosion(enemy.rect.centerx, enemy.rect.centery, 80)
                            all_sprites.add(exp)
                            
                            if isinstance(enemy, BossBaal) and not door_spawned:
                                # La puerta se crea a la derecha y la llave dorada aparece claramente delante de ella
                                door = DoorObject(WIDTH - 170, GROUND_Y)
                                all_sprites.add(door)
                                doors_group.add(door)

                                key = KeyItem(WIDTH - 230, GROUND_Y - 40)
                                all_sprites.add(key)
                                keys_group.add(key)

                                door_spawned = True

                            enemy.kill()
                    else:
                        play_small_bug_death()
                        exp = NoteExplosion(enemy.rect.centerx, enemy.rect.centery, (255, 100, 0))
                        all_sprites.add(exp)
                        enemy.kill()
                    p.kill()

            for key in keys_group:
                if timmy.rect.colliderect(key.rect):
                    timmy.has_key = True
                    play_key_pickup_sound()
                    key.kill()
                    for door in doors_group:
                        door.open_door()

            for door in doors_group:
                if door.is_open and timmy.rect.colliderect(door.rect):
                    victory = True
                    play_fur_elise_victory()

            for fb in fireballs:
                if timmy.rect.colliderect(fb.rect):
                    timmy.lives -= 1
                    play_tone(100, 0.2, 0.4)
                    if timmy.lives <= 0:
                        game_over = True
                        play_defeat_sound()
                    fb.kill()

            for enemy in enemies:
                if timmy.rect.colliderect(enemy.rect):
                    timmy.lives -= 1
                    play_tone(110, 0.2, 0.4)
                    if timmy.lives <= 0:
                        game_over = True
                        play_defeat_sound()
                    timmy.rect.x -= 20

        bg_art.draw(screen, scroll_offset, boss_spawned)

        pygame.draw.rect(screen, COLOR_STAFF, (0, GROUND_Y, WIDTH, HEIGHT - GROUND_Y))
        for i in range(5):
            line_y = STAFF_BOTTOM_Y - i * STAFF_STEP
            pygame.draw.line(screen, (190, 190, 210), (0, line_y), (WIDTH, line_y), 2)

        all_sprites.draw(screen)
        keys_group.draw(screen)
        doors_group.draw(screen)
        silence_bombs.draw(screen)
        fireballs.draw(screen)

        remaining_sec = max(0, int(GAME_DURATION_SEC - ((pygame.time.get_ticks() - start_ticks) / 1000))) if not boss_spawned else 0

        txt_lives = font_hud.render(f"Vidas: {timmy.lives}", True, COLOR_HUD)
        txt_score = font_hud.render(f"Puntuación: {int(timmy.score_times)}", True, COLOR_HUD)
        txt_sharps = font_hud.render(f"Sostenidos (#): {timmy.ammo_sharps}", True, COLOR_SHARP)
        txt_flats = font_hud.render(f"Bemoles (b): {timmy.ammo_flats}", True, COLOR_FLAT)
        txt_level = font_hud.render(f"NIVEL 2: EL VIOLÍN (Jefe en: {remaining_sec}s)" if not boss_spawned else "NIVEL 2: ¡JEFE BAAL!", True, COLOR_HUD)

        screen.blit(txt_lives, (20, 15))
        screen.blit(txt_score, (150, 15))
        screen.blit(txt_sharps, (320, 15))
        screen.blit(txt_flats, (520, 15))
        screen.blit(txt_level, (720, 15))

        if boss_spawned and boss_ref and boss_ref.alive():
            pygame.draw.rect(screen, (50, 50, 50), (WIDTH - 300, 50, 260, 20), border_radius=5)
            hp_ratio = max(0, boss_ref.hp / boss_ref.max_hp)
            pygame.draw.rect(screen, (180, 20, 180), (WIDTH - 300, 50, int(260 * hp_ratio), 20), border_radius=5)
            boss_txt = font_boss.render("BAAL", True, (255, 255, 255))
            screen.blit(boss_txt, (WIDTH - 200, 20))

        if baal_title_timer > 0:
            baal_surf = font_huge_baal.render("¡BAAL HA DESPERTADO!", True, (200, 40, 220))
            screen.blit(baal_surf, (WIDTH//2 - baal_surf.get_width()//2, HEIGHT//3))

            screen.blit(overlay, (0, 0))
            pause_txt = font_big.render("JUEGO EN PAUSA", True, (255, 255, 255))
            screen.blit(pause_txt, (WIDTH//2 - pause_txt.get_width()//2, HEIGHT//2 - 30))

        if game_over or victory:
            overlay = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA)
            overlay.fill((0, 0, 0, 180))
            screen.blit(overlay, (0, 0))
            if victory:
                msg = font_big.render("¡NIVEL 3 DESBLOQUEADO!", True, (100, 255, 100))
            else:
                msg = font_big.render("¡GAME OVER!", True, (255, 100, 100))
            screen.blit(msg, (WIDTH//2 - msg.get_width()//2, HEIGHT//2 - 60))
            
            sub_msg = font_btn.render("Presiona 'R' para Reiniciar o 'Q' para Salir", True, (255, 255, 255))
            screen.blit(sub_msg, (WIDTH//2 - sub_msg.get_width()//2, HEIGHT//2 + 10))

        pygame.display.flip()

    return "QUIT"

def main():
    while True:
        res = run_level()
        if res == "QUIT":

if __name__ == "__main__":
