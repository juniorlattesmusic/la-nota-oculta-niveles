def play_tone(freq, duration=0.4, volume=0.3):
    if not AUDIO_AVAILABLE: return
    try:
        n_samples = int(AUDIO_FREQ * duration)
        buf = array.array('h', [0] * n_samples)
        max_amp = 32767 * volume
        for i in range(n_samples):
            t = float(i) / AUDIO_FREQ
            envelope = max(0.0, 1.0 - (i / n_samples))
            val = int(math.sin(2.0 * math.pi * freq * t) * max_amp * envelope)
            buf[i] = val
        sound = pygame.mixer.Sound(buffer=buf)
        sound.play()
    except Exception: pass

        self.rect = self.image.get_rect()

    def update(self):
        alive = False
        self.image.fill((0, 0, 0, 0))
        for p in self.particles:
            if p[4] > 0:
                p[0] += p[2]
                p[1] += p[3]
                p[4] -= 1

        self.wobble_timer += 0.12
        dx = self.target.rect.x - self.rect.x
        dy = self.target.rect.y - self.rect.y
        dist = math.hypot(dx, dy)
        if dist != 0:
            self.rect.x += (dx / dist) * self.base_speed
            self.rect.y += (dy / dist) * (self.base_speed * 0.4) + math.sin(self.wobble_timer) * 1.5
        if self.rect.bottom >= GROUND_Y:
            self.rect.bottom = GROUND_Y
        if self.rect.right < 0:
            self.kill()

class GiantSpiderEnemy(pygame.sprite.Sprite):
    def __init__(self, enemy_bullets_group):
        super().__init__()
        self.image = pygame.Surface((90, 70), pygame.SRCALPHA)
        self.rect = self.image.get_rect(x=WIDTH + random.randint(40, 120), y=GROUND_Y - 70)
        self.hp = 12
        self.max_hp = 12
        self.enemy_bullets = enemy_bullets_group
        self.shoot_timer = 0
        self._draw_giant_spider()

    def _draw_giant_spider(self):
        self.image.fill((0, 0, 0, 0))
        pygame.draw.ellipse(self.image, COLOR_BIG_SPIDER, (15, 12, 60, 44))
        for dx in range(30, 55, 4):
            pygame.draw.polygon(self.image, (255, 255, 255), [(dx, 48), (dx+2, 54), (dx+4, 48)])
        for ox in [30, 40, 50, 60]:
            pygame.draw.circle(self.image, (255, 0, 0), (ox, 22), 4)
        pygame.draw.lines(self.image, COLOR_BIG_SPIDER, False, [(20, 25), (4, 5), (0, 50)], 5)
        pygame.draw.lines(self.image, COLOR_BIG_SPIDER, False, [(70, 25), (86, 5), (90, 50)], 5)
        pygame.draw.rect(self.image, (100, 100, 100), (20, 0, 50, 6))
        hp_w = int(50 * (self.hp / self.max_hp))
        pygame.draw.rect(self.image, (200, 0, 0), (20, 0, max(0, hp_w), 6))

    def update(self):
        self.rect.x -= 1.05
        self.shoot_timer += 1
        if self.shoot_timer >= 220:
            self.shoot_timer = 0
            for angle in [-0.2, 0.0, 0.2]:
                vx = -3.5 * math.cos(angle)
                vy = 3.5 * math.sin(angle)
                bullet = EnemyBullet(self.rect.left, self.rect.centery, vx, vy)
                self.enemy_bullets.add(bullet)
        if self.rect.right < 0:
            self.kill()

        self.attack_timer += 1
        if self.attack_timer % 6 == 0:
            self.spiral_angle += 0.5
        if self.attack_timer % 24 == 0:
            play_boss_ugly_sound()
            for factor in [1, -1, 0]:
                vx = math.cos(self.spiral_angle * factor if factor != 0 else self.spiral_angle) * 6.0
                vy = math.sin(self.spiral_angle * factor if factor != 0 else self.spiral_angle) * 6.0
                b = EnemyBullet(self.rect.centerx - 40, self.rect.centery, vx, vy)
                self.enemy_bullets.add(b)

def run_level(level_idx):
    timmy = Timmy()
    timmy.set_level_color(level_idx)
    sun_clouds = SunAndClouds()
    bg_art = BackgroundArt()
    all_sprites = pygame.sprite.Group(timmy)
    projectiles = pygame.sprite.Group()
    enemy_bullets = pygame.sprite.Group()
    items = pygame.sprite.Group()
    enemies = pygame.sprite.Group()
    clef_boxes = pygame.sprite.Group()
    silence_bombs = pygame.sprite.Group()
    explosions = pygame.sprite.Group()

    void_gaps = []
    if level_idx >= 1:
        for _ in range(2):
            gx = random.randint(350, WIDTH - 200)
            void_gaps.append((gx, gx + 80))

    start_ticks = pygame.time.get_ticks()
    boss_spawned = False
    boss_ref = None
    game_over = False
    victory = False
    is_paused = False

    spawn_enemy_timer = 0
    spawn_giant_timer = 0
    spawn_item_timer = 0
    spawn_bomb_timer = 0
    song_idx = level_idx % len(SONGS)
    note_in_song_idx = 0

    font_hud = pygame.font.Font(None, 26)
    font_boss = pygame.font.Font(None, 34)
    font_big = pygame.font.Font(None, 54)
    font_btn = pygame.font.Font(None, 32)

    running = True
    while running:
        clock.tick(FPS)
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                return "QUIT"
            elif event.type == pygame.KEYDOWN:
                if event.key == pygame.K_ESCAPE:
                    return "QUIT"
                if event.key == pygame.K_p:
                    is_paused = not is_paused
                if game_over or victory:
                    if event.key == pygame.K_r:
                        return "RESTART"
                    elif event.key == pygame.K_q:
                        return "QUIT"
                if not game_over and not victory and not is_paused:
                    if event.key in (pygame.K_SPACE, pygame.K_w, pygame.K_UP):
                        timmy.jump()
                    elif event.key in (pygame.K_s, pygame.K_DOWN):
                        timmy.crouch(True)
                    elif event.key in (pygame.K_j, pygame.K_z):
                        if timmy.ammo_sharps > 0:
                            p = Projectile(timmy.rect.right, timmy.rect.centery, "SHARP", timmy.weapon_level)
                            all_sprites.add(p)
                            projectiles.add(p)
                            timmy.ammo_sharps -= 1
                            play_pew()
                    elif event.key in (pygame.K_k, pygame.K_x):
                        if timmy.ammo_flats > 0:
                            p = Projectile(timmy.rect.right, timmy.rect.centery, "FLAT_GRANADE")
                            all_sprites.add(p)
                            projectiles.add(p)
                            timmy.ammo_flats -= 1
                            play_pew()
            elif event.type == pygame.KEYUP:
                if event.key in (pygame.K_s, pygame.K_DOWN):
                    timmy.crouch(False)

        if not game_over and not victory and not is_paused:
            elapsed_sec = (pygame.time.get_ticks() - start_ticks) / 1000
            difficulty_factor = min(1.0, elapsed_sec / GAME_DURATION_SEC)

            if not boss_spawned:
                spawn_enemy_timer += 1
                spawn_giant_timer += 1
                spawn_item_timer += 1
                spawn_bomb_timer += 1

                if spawn_enemy_timer >= max(20, int(50 - (difficulty_factor * 20))):
                    spider = SpiderEnemy(timmy, difficulty_factor)
                    all_sprites.add(spider)
                    enemies.add(spider)
                    spawn_enemy_timer = 0

                if spawn_giant_timer >= 450:
                    g_spider = GiantSpiderEnemy(enemy_bullets)
                    all_sprites.add(g_spider)
                    enemies.add(g_spider)
                    spawn_giant_timer = 0

                bomb_interval = max(90, int(220 - (difficulty_factor * 80)))
                if spawn_bomb_timer >= bomb_interval:
                    s_bomb = SilenceBomb(difficulty_factor)
                    all_sprites.add(s_bomb)
                    silence_bombs.add(s_bomb)
                    spawn_bomb_timer = 0

                if spawn_item_timer >= 50:
                    current_song = SONGS[song_idx]
                    note_key, figure_type = current_song[note_in_song_idx]
                    note_in_song_idx += 1
                    if note_in_song_idx >= len(current_song):
                        note_in_song_idx = 0

                    note = ItemNote(note_key, figure_type, difficulty_factor)
                    all_sprites.add(note)
                    items.add(note)

                    if random.random() < 0.35:
                        clef = ClefBox()
                        all_sprites.add(clef)
                        clef_boxes.add(clef)
                    spawn_item_timer = 0

                if elapsed_sec >= GAME_DURATION_SEC and not boss_spawned:
                    boss_spawned = True
                    boss_ref = BossMonstruoFinal(enemy_bullets)
                    all_sprites.add(boss_ref)
                    enemies.add(boss_ref)

            all_sprites.update()
            silence_bombs.update()
            enemy_bullets.update()
            explosions.update()

            in_void = False
            for start_x, end_x in void_gaps:
                if start_x <= timmy.rect.centerx <= end_x and timmy.rect.bottom >= GROUND_Y - 5:
                    in_void = True
                    break
            if in_void:
                timmy.vel_y += 1.8
                timmy.rect.y += timmy.vel_y
                if timmy.rect.top > HEIGHT:
                    timmy.lives = 0
                    game_over = True
                    play_defeat_sound()

            for note in items:
                if timmy.rect.colliderect(note.rect):
                    play_tone(note.freq, note.sound_duration, 0.4)
                    timmy.score_times += note.value
                    if note.type == "REDONDA":
                        timmy.lives = min(25, timmy.lives + 3)
                    elif note.type == "BLANCA":
                        timmy.lives = min(25, timmy.lives + 1)

                    timmy.evolve_weapon_check()
                    exp = NoteExplosion(note.rect.centerx, note.rect.centery, note.color)
                    all_sprites.add(exp)
                    note.kill()

            for clef in clef_boxes:
                if timmy.rect.colliderect(clef.rect):
                    timmy.ammo_sharps += 20
                    timmy.ammo_flats += 10
                    play_tone(440, 0.2, 0.3)
                    clef.kill()

            for bomb in silence_bombs:
                if timmy.rect.colliderect(bomb.rect):
                    timmy.lives -= 3
                    play_tone(100, 0.3, 0.4)
                    if timmy.lives <= 0:
                        game_over = True
                        play_defeat_sound()
                    bomb.kill()

            for p in projectiles:
                for enemy in pygame.sprite.spritecollide(p, enemies, False):
                    sun_clouds.trigger_surprise()
                    if isinstance(enemy, GiantSpiderEnemy) or isinstance(enemy, BossMonstruoFinal):
                        enemy.hp -= 1
                        if enemy.hp <= 0:
                            exp = MassiveExplosion(enemy.rect.centerx, enemy.rect.centery, 80)
                            all_sprites.add(exp)
                            enemy.kill()
                            if isinstance(enemy, BossMonstruoFinal):
                                victory = True
                                play_victory_trumpet_fanfare()
                    else:
                        exp = NoteExplosion(enemy.rect.centerx, enemy.rect.centery, (255, 100, 0))
                        all_sprites.add(exp)
                        enemy.kill()
                    p.kill()

            for bullet in enemy_bullets:
                if timmy.rect.colliderect(bullet.rect):
                    timmy.lives -= 1
                    play_tone(120, 0.2, 0.4)
                    if timmy.lives <= 0:
                        game_over = True
                        play_defeat_sound()
                    bullet.kill()

            for enemy in enemies:
                if timmy.rect.colliderect(enemy.rect):
                    timmy.lives -= 1
                    play_tone(110, 0.2, 0.4)
                    if timmy.lives <= 0:
                        game_over = True
                        play_defeat_sound()
                    timmy.rect.x -= 20

        screen.fill(COLOR_BG)
        bg_art.draw(screen, level_idx + 1)
        sun_clouds.draw(screen, level_idx + 1)

        pygame.draw.rect(screen, COLOR_STAFF, (0, GROUND_Y, WIDTH, HEIGHT - GROUND_Y))
        for line_y in range(STAFF_BOTTOM_Y, STAFF_BOTTOM_Y - STAFF_STEP * 5, -STAFF_STEP):
            pygame.draw.line(screen, (180, 180, 190), (0, line_y), (WIDTH, line_y), 2)

        for sx, ex in void_gaps:
            pygame.draw.rect(screen, COLOR_BG, (sx, GROUND_Y - 5, ex - sx, HEIGHT - GROUND_Y + 5))

        all_sprites.draw(screen)
        silence_bombs.draw(screen)
        enemy_bullets.draw(screen)

        txt_lives = font_hud.render(f"Vidas: {timmy.lives}", True, COLOR_HUD)
        txt_score = font_hud.render(f"Puntuación: {int(timmy.score_times)}", True, COLOR_HUD)
        txt_sharps = font_hud.render(f"Sostenidos (#): {timmy.ammo_sharps}", True, COLOR_SHARP)
        txt_flats = font_hud.render(f"Bemoles (b): {timmy.ammo_flats}", True, COLOR_FLAT)
        txt_level = font_hud.render(f"Nivel: {level_idx + 1}", True, COLOR_HUD)

        screen.blit(txt_lives, (20, 15))
        screen.blit(txt_score, (150, 15))
        screen.blit(txt_sharps, (320, 15))
        screen.blit(txt_flats, (520, 15))
        screen.blit(txt_level, (720, 15))

        if boss_spawned and boss_ref and boss_ref.alive():
            pygame.draw.rect(screen, (50, 50, 50), (WIDTH - 300, 50, 260, 20), border_radius=5)
            hp_ratio = max(0, boss_ref.hp / boss_ref.max_hp)
            pygame.draw.rect(screen, (200, 30, 30), (WIDTH - 300, 50, int(260 * hp_ratio), 20), border_radius=5)
            boss_txt = font_boss.render("¡MONSTRUO FINAL!", True, (255, 255, 255))
            screen.blit(boss_txt, (WIDTH - 280, 20))

        if is_paused:
            overlay = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA)
            overlay.fill((0, 0, 0, 150))
            screen.blit(overlay, (0, 0))
            pause_txt = font_big.render("JUEGO EN PAUSA", True, (255, 255, 255))
            screen.blit(pause_txt, (WIDTH//2 - pause_txt.get_width()//2, HEIGHT//2 - 30))

        if game_over or victory:
            overlay = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA)
            overlay.fill((0, 0, 0, 180))
            screen.blit(overlay, (0, 0))
            if victory:
                msg = font_big.render("¡VICTORIA! ¡NIVEL SUPERADO!", True, (100, 255, 100))
            else:
                msg = font_big.render("¡GAME OVER - JUEGO TERMINADO!", True, (255, 100, 100))
            screen.blit(msg, (WIDTH//2 - msg.get_width()//2, HEIGHT//2 - 60))
            
            sub_msg = font_btn.render("Presiona 'R' para Reiniciar o 'Q' para Salir", True, (255, 255, 255))
            screen.blit(sub_msg, (WIDTH//2 - sub_msg.get_width()//2, HEIGHT//2 + 10))

        pygame.display.flip()

    return "QUIT"

def main():
    current_level = 0
    while current_level < len(LEVEL_COLORS):
        result = run_level(current_level)
        if result == "QUIT":
            break
        elif result == "RESTART":
            continue
        current_level += 1
    pygame.quit()
    sys.exit()

if __name__ == "__main__":
    main()
